cmake_minimum_required(VERSION 3.28)
project(difftest_report_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
if(NOT CMAKE_GENERATOR MATCHES "Ninja")
    message(FATAL_ERROR "difftest_report_test requires the Ninja generator")
endif()
get_filename_component(DIFFTEST_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../difftest ABSOLUTE)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    file(GLOB std_candidates "/usr/include/c++/*/bits/std.cc")
    if(NOT std_candidates)
        message(FATAL_ERROR "找不到 GCC 的 bits/std.cc")
    endif()
    list(SORT std_candidates)
    list(GET std_candidates -1 std_source)
    file(COPY ${std_source} DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
    set(std_file ${CMAKE_CURRENT_BINARY_DIR}/std.cc)
    set(std_flags)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(std_source /usr/share/libc++/v1/std.cppm)
    if(NOT EXISTS ${std_source})
        message(FATAL_ERROR "找不到 libc++ 的 std.cppm")
    endif()
    file(COPY ${std_source} DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
    set(std_file ${CMAKE_CURRENT_BINARY_DIR}/std.cppm)
    set(std_flags -stdlib=libc++)
else()
    message(FATAL_ERROR "仅支持 GCC 或 Clang 的 std 模块")
endif()
add_executable(difftest-report-test main.cpp)
target_sources(difftest-report-test PRIVATE
    FILE_SET CXX_MODULES
    BASE_DIRS ${DIFFTEST_DIR} ${CMAKE_CURRENT_BINARY_DIR}
    FILES ${DIFFTEST_DIR}/DifftestReport.ixx ${std_file}
)
target_compile_options(difftest-report-test PRIVATE ${std_flags} -Wall -Wextra -Werror -D_GLIBCXX_ASSERTIONS)
target_link_options(difftest-report-test PRIVATE ${std_flags})
set_target_properties(difftest-report-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/build)
